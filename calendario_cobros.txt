function CalendarioCobros(_ref_cc){
  let clientes=_ref_cc.clientes||[];
  const clientesActivos=useMemo(()=>(clientes||[]).filter(c=>c.activo!==false).slice().sort((a,b)=>a.nombre.localeCompare(b.nombre)),[clientes]);
  const _ccS1=useState(''),clienteSel=_ccS1[0],setClienteSel=_ccS1[1];
  useEffect(()=>{if(!clienteSel&&clientesActivos.length>0)setClienteSel(clientesActivos[0].nombre);},[clientesActivos]);
  const _ccS2=useState([]),envios=_ccS2[0],setEnvios=_ccS2[1];
  const _ccS3=useState(true),cargando=_ccS3[0],setCargando=_ccS3[1];
  const _ccS4=useState(()=>{const d=new Date();d.setDate(1);d.setHours(0,0,0,0);return d;}),mesVisible=_ccS4[0],setMesVisible=_ccS4[1];
  const _ccS5=useState(()=>new Set()),diasSeleccionados=_ccS5[0],setDiasSeleccionados=_ccS5[1];
  const _ccS6=useState(false),generando=_ccS6[0],setGenerando=_ccS6[1];

  useEffect(()=>{
    if(!clienteSel)return;
    let cancelado=false;
    async function cargar(){
      setCargando(true);
      try{
        let acumulado=[];
        let desde=0;
        const bloque=1000;
        while(true){
          const{data,error}=await db.from('envios').select('codigo,fecha,estado,cliente,comuna,peso,valor_siniestro,updated_at')
            .eq('cliente',clienteSel)
            .order('fecha',{ascending:false}).range(desde,desde+bloque-1);
          if(error||!data||data.length===0)break;
          acumulado=acumulado.concat(data);
          if(data.length<bloque)break;
          desde+=bloque;
          if(cancelado)return;
        }
        if(!cancelado)setEnvios(acumulado);
      }catch(e){console.warn('Error cargando calendario de cobros:',e.message);}
      if(!cancelado){setCargando(false);setDiasSeleccionados(new Set());}
    }
    cargar();
    return()=>{cancelado=true;};
  },[clienteSel]);

  function localDateStr(d){
    const dt=d?new Date(d):new Date();
    return dt.getFullYear()+'-'+String(dt.getMonth()+1).padStart(2,'0')+'-'+String(dt.getDate()).padStart(2,'0');
  }
  function fechaEnvio(e){return e.fecha||localDateStr(e.updated_at);}

  const conteoPorDia=useMemo(()=>{
    const mapa={};
    envios.forEach(e=>{
      const key=fechaEnvio(e);
      if(!mapa[key])mapa[key]={total:0,entregado:0};
      mapa[key].total++;
      if(e.estado==='entregado')mapa[key].entregado++;
    });
    return mapa;
  },[envios]);

  const semanas=useMemo(()=>{
    const anio=mesVisible.getFullYear();
    const mes=mesVisible.getMonth();
    const primerDiaMes=new Date(anio,mes,1);
    const ultimoDiaMes=new Date(anio,mes+1,0);
    const inicioGrilla=new Date(primerDiaMes);
    inicioGrilla.setDate(primerDiaMes.getDate()-((primerDiaMes.getDay()+6)%7));
    const finGrilla=new Date(ultimoDiaMes);
    finGrilla.setDate(ultimoDiaMes.getDate()+(7-((ultimoDiaMes.getDay()+6)%7)-1));
    const semanasArr=[];
    let cursor=new Date(inicioGrilla);
    const hoyStr=localDateStr(new Date());
    while(cursor<=finGrilla){
      const semana=[];
      for(let i=0;i<7;i++){
        const fechaStr=localDateStr(cursor);
        const d=conteoPorDia[fechaStr]||{total:0,entregado:0};
        semana.push({fecha:new Date(cursor),fechaStr,enMes:cursor.getMonth()===mes,esHoy:fechaStr===hoyStr,total:d.total,entregado:d.entregado});
        cursor.setDate(cursor.getDate()+1);
      }
      semanasArr.push(semana);
    }
    return semanasArr;
  },[mesVisible,conteoPorDia]);

  const totalMes=useMemo(()=>{
    const anio=mesVisible.getFullYear();
    const mes=mesVisible.getMonth();
    let total=0,entregado=0,dias=0;
    Object.keys(conteoPorDia).forEach(k=>{
      const d=new Date(k+'T00:00:00');
      if(d.getFullYear()===anio&&d.getMonth()===mes){
        total+=conteoPorDia[k].total;
        entregado+=conteoPorDia[k].entregado;
        if(conteoPorDia[k].total>0)dias++;
      }
    });
    return{total,entregado,dias,efectividad:total>0?entregado/total:0};
  },[conteoPorDia,mesVisible]);

  function colorEfectividad(total,entregado){
    if(total===0)return{bg:'var(--cream)',border:'var(--border)',texto:'var(--text-soft)'};
    const ratio=entregado/total;
    const c=barColor(ratio);
    return{bg:c+'1a',border:c+'55',texto:c};
  }

  function toggleDia(fechaStr){
    setDiasSeleccionados(prev=>{
      const n=new Set(prev);
      if(n.has(fechaStr))n.delete(fechaStr);else n.add(fechaStr);
      return n;
    });
  }

  function seleccionarMes(){
    setDiasSeleccionados(prev=>{
      const n=new Set(prev);
      const anio=mesVisible.getFullYear();
      const mes=mesVisible.getMonth();
      Object.keys(conteoPorDia).forEach(k=>{
        const d=new Date(k+'T00:00:00');
        if(d.getFullYear()===anio&&d.getMonth()===mes&&conteoPorDia[k].total>0)n.add(k);
      });
      return n;
    });
  }

  function seleccionarUltimos15(){
    setDiasSeleccionados(prev=>{
      const n=new Set(prev);
      const diasConDatos=Object.keys(conteoPorDia).filter(k=>conteoPorDia[k].total>0).sort();
      diasConDatos.slice(-15).forEach(k=>n.add(k));
      return n;
    });
  }

  function limpiarSeleccion(){setDiasSeleccionados(new Set());}

  const resumenSeleccion=useMemo(()=>{
    let total=0,entregado=0;
    envios.forEach(e=>{
      if(diasSeleccionados.has(fechaEnvio(e))){
        total++;
        if(e.estado==='entregado')entregado++;
      }
    });
    return{total,entregado,efectividad:total>0?entregado/total:0};
  },[diasSeleccionados,envios]);

  async function generarRecibo(){
    if(diasSeleccionados.size===0)return;
    setGenerando(true);
    try{
      const enviosSel=envios.filter(e=>diasSeleccionados.has(fechaEnvio(e)));
      const cliData=(clientes||[]).find(c=>c.nombre===clienteSel)||{};
      const diasOrdenados=Array.from(diasSeleccionados).sort();
      let periodoLabel;
      if(diasOrdenados.length===1){
        periodoLabel=diasOrdenados[0];
      }else{
        const fmtCorto=f=>{const d=new Date(f+'T00:00:00');return d.getDate()+'/'+(d.getMonth()+1);};
        periodoLabel=diasOrdenados.length+' días seleccionados ('+diasOrdenados.map(fmtCorto).join(', ')+')';
      }
      await generarReciboCobro(clienteSel,enviosSel,periodoLabel,cliData);
    }finally{
      setGenerando(false);
    }
  }

  const nombresMes=['Enero','Febrero','Marzo','Abril','Mayo','Junio','Julio','Agosto','Septiembre','Octubre','Noviembre','Diciembre'];
  const diasSemana=['L','M','M','J','V','S','D'];
  const esMesActual=mesVisible.getFullYear()===new Date().getFullYear()&&mesVisible.getMonth()===new Date().getMonth();

  return React.createElement('div',null,
    React.createElement('div',{className:'section-head'},React.createElement('div',{className:'section-title'},'Calendario ',React.createElement('span',null,'de Cobros'))),
    React.createElement('div',{className:'info-banner'},'🔒 Solo visible para Super Admin. Marca los días que quieras cobrarle a un cliente (no tienen que ser seguidos) y genera un Recibo de Cobro consolidado con esos días.'),
    React.createElement('div',{style:{display:'flex',gap:10,alignItems:'center',marginBottom:16,flexWrap:'wrap'}},
      React.createElement('div',{style:{fontFamily:'Bebas Neue',fontSize:13,letterSpacing:2,color:'var(--text-soft)'}},'CLIENTE:'),
      React.createElement('select',{value:clienteSel,onChange:e=>setClienteSel(e.target.value),style:{padding:'8px 14px',borderRadius:8,border:'1px solid var(--gold)',fontSize:13,fontWeight:600,color:'var(--dark)',minWidth:220}},
        clientesActivos.map(c=>React.createElement('option',{key:c.nombre,value:c.nombre},c.nombre))
      )
    ),
    React.createElement('div',{className:'panel'},
      React.createElement('div',{style:{display:'flex',alignItems:'center',justifyContent:'space-between',padding:'0 4px',marginBottom:14}},
        React.createElement('button',{onClick:()=>setMesVisible(m=>{const n=new Date(m);n.setMonth(n.getMonth()-1);return n;}),style:{background:'var(--cream)',border:'1px solid var(--border)',color:'var(--gold)',fontSize:18,width:38,height:38,borderRadius:10,cursor:'pointer'}},'‹'),
        React.createElement('div',{style:{fontFamily:'Bebas Neue',fontSize:18,letterSpacing:1.5,color:'var(--dark)'}},nombresMes[mesVisible.getMonth()]+' '+mesVisible.getFullYear()),
        React.createElement('button',{onClick:()=>setMesVisible(m=>{const n=new Date(m);n.setMonth(n.getMonth()+1);return n;}),disabled:esMesActual,style:{background:'var(--cream)',border:'1px solid var(--border)',color:esMesActual?'rgba(200,168,75,0.3)':'var(--gold)',fontSize:18,width:38,height:38,borderRadius:10,cursor:esMesActual?'default':'pointer'}},'›')
      ),
      React.createElement('div',{style:{display:'flex',justifyContent:'space-around',textAlign:'center',background:'rgba(200,168,75,0.08)',border:'1px solid rgba(200,168,75,0.3)',borderRadius:12,padding:'14px 18px',marginBottom:16}},
        React.createElement('div',null,
          React.createElement('div',{style:{fontFamily:'Bebas Neue',fontSize:24,color:'var(--gold)'}},totalMes.entregado+'/'+totalMes.total),
          React.createElement('div',{style:{fontSize:10,color:'var(--text-soft)',letterSpacing:1,textTransform:'uppercase',marginTop:2}},'Entregadas / mes')
        ),
        React.createElement('div',null,
          React.createElement('div',{style:{fontFamily:'Bebas Neue',fontSize:24,color:barColor(totalMes.efectividad)}},pct(totalMes.efectividad)),
          React.createElement('div',{style:{fontSize:10,color:'var(--text-soft)',letterSpacing:1,textTransform:'uppercase',marginTop:2}},'Efectividad')
        ),
        React.createElement('div',null,
          React.createElement('div',{style:{fontFamily:'Bebas Neue',fontSize:24,color:'var(--dark)'}},totalMes.dias),
          React.createElement('div',{style:{fontSize:10,color:'var(--text-soft)',letterSpacing:1,textTransform:'uppercase',marginTop:2}},'Días con movimiento')
        )
      ),
      React.createElement('div',{style:{display:'flex',gap:8,marginBottom:14,flexWrap:'wrap'}},
        React.createElement('button',{onClick:seleccionarMes,style:{padding:'6px 14px',borderRadius:20,border:'1px solid var(--gold)',background:'rgba(200,168,75,0.08)',color:'var(--gold)',fontWeight:700,fontSize:11,cursor:'pointer'}},'Seleccionar mes completo'),
        React.createElement('button',{onClick:seleccionarUltimos15,style:{padding:'6px 14px',borderRadius:20,border:'1px solid var(--gold)',background:'rgba(200,168,75,0.08)',color:'var(--gold)',fontWeight:700,fontSize:11,cursor:'pointer'}},'Últimos 15 días'),
        diasSeleccionados.size>0&&React.createElement('button',{onClick:limpiarSeleccion,style:{padding:'6px 14px',borderRadius:20,border:'1px solid var(--border)',background:'#fff',color:'var(--text-soft)',fontWeight:700,fontSize:11,cursor:'pointer'}},'✕ Limpiar selección ('+diasSeleccionados.size+')')
      ),
      React.createElement('div',{style:{display:'grid',gridTemplateColumns:'repeat(7,1fr)',gap:4,marginBottom:4}},
        diasSemana.map((d,i)=>React.createElement('div',{key:i,style:{textAlign:'center',fontSize:10,color:'var(--text-soft)',fontWeight:700,letterSpacing:1}},d))
      ),
      semanas.map((semana,wi)=>
        React.createElement('div',{key:wi,style:{display:'grid',gridTemplateColumns:'repeat(7,1fr)',gap:4,marginBottom:4}},
          semana.map((dia,di)=>{
            const col=colorEfectividad(dia.total,dia.entregado);
            const sel=diasSeleccionados.has(dia.fechaStr);
            return React.createElement('div',{
              key:di,
              onClick:dia.total>0?()=>toggleDia(dia.fechaStr):undefined,
              style:{
                aspectRatio:'1',borderRadius:8,
                background:sel?'var(--gold)':col.bg,
                border:sel?'2px solid var(--dark)':(dia.esHoy?'2px solid var(--gold)':'1px solid '+col.border),
                opacity:dia.enMes?1:0.3,
                display:'flex',flexDirection:'column',alignItems:'center',justifyContent:'center',
                minHeight:52,cursor:dia.total>0?'pointer':'default',position:'relative'
              }
            },
              sel&&React.createElement('div',{style:{position:'absolute',top:3,right:4,fontSize:10,color:'#fff'}},'✓'),
              React.createElement('div',{style:{fontSize:10,color:sel?'#fff':(dia.enMes?'var(--text-soft)':'rgba(0,0,0,0.2)'),fontWeight:600}},dia.fecha.getDate()),
              dia.total>0&&React.createElement('div',{style:{fontFamily:'Bebas Neue',fontSize:13,color:sel?'#fff':(dia.enMes?col.texto:'rgba(0,0,0,0.2)'),lineHeight:1,marginTop:2}},dia.entregado+'/'+dia.total)
            );
          })
        )
      ),
      cargando&&React.createElement('div',{style:{textAlign:'center',padding:20,color:'var(--text-soft)',fontSize:12}},'Cargando historial de '+clienteSel+'...'),
      !cargando&&envios.length===0&&React.createElement('div',{style:{textAlign:'center',padding:'20px 30px',color:'var(--text-soft)',fontSize:12}},'Sin envíos registrados para este cliente.')
    ),
    diasSeleccionados.size>0&&React.createElement('div',{style:{position:'sticky',bottom:16,marginTop:16,background:'#fff',border:'2px solid var(--gold)',borderRadius:14,padding:'16px 20px',display:'flex',alignItems:'center',justifyContent:'space-between',gap:16,flexWrap:'wrap',boxShadow:'0 6px 20px rgba(0,0,0,0.12)'}},
      React.createElement('div',{style:{display:'flex',gap:20,flexWrap:'wrap'}},
        React.createElement('div',null,
          React.createElement('div',{style:{fontFamily:'Bebas Neue',fontSize:20,color:'var(--dark)'}},diasSeleccionados.size),
          React.createElement('div',{style:{fontSize:9,color:'var(--text-soft)',letterSpacing:1,textTransform:'uppercase'}},'Días seleccionados')
        ),
        React.createElement('div',null,
          React.createElement('div',{style:{fontFamily:'Bebas Neue',fontSize:20,color:'var(--gold)'}},resumenSeleccion.entregado+'/'+resumenSeleccion.total),
          React.createElement('div',{style:{fontSize:9,color:'var(--text-soft)',letterSpacing:1,textTransform:'uppercase'}},'Entregados / Recibidos')
        ),
        React.createElement('div',null,
          React.createElement('div',{style:{fontFamily:'Bebas Neue',fontSize:20,color:barColor(resumenSeleccion.efectividad)}},pct(resumenSeleccion.efectividad)),
          React.createElement('div',{style:{fontSize:9,color:'var(--text-soft)',letterSpacing:1,textTransform:'uppercase'}},'Efectividad')
        )
      ),
      React.createElement('button',{onClick:generarRecibo,disabled:generando,style:{padding:'12px 24px',borderRadius:10,border:'none',background:generando?'var(--border)':'linear-gradient(135deg,var(--gold),#a87d2a)',color:generando?'var(--text-soft)':'#1a1d13',fontWeight:700,fontSize:13,cursor:generando?'default':'pointer',whiteSpace:'nowrap'}},generando?'Generando...':'📄 Generar Recibo de Cobro')
    )
  );
}
